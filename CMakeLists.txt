cmake_minimum_required(VERSION 3.25)
project(gfx-research-shadows VERSION 0.16.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(GFX_RESEARCH_BASE_LOCAL_PATH "" CACHE PATH "Optional local gfx-research-base checkout")
set(GFX_RESEARCH_BASE_GIT_TAG "a419073ff7c5fd781ef52a4ff9fcdb0c67c5275c" CACHE STRING "Pinned base revision")
set(GFX_RESEARCH_BASE_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
if(GFX_RESEARCH_BASE_LOCAL_PATH)
    add_subdirectory("${GFX_RESEARCH_BASE_LOCAL_PATH}" "${CMAKE_BINARY_DIR}/gfx-research-base")
else()
    include(FetchContent)
    FetchContent_Declare(gfx_research_base
        GIT_REPOSITORY https://github.com/Terrasil/gfx-research-base.git
        GIT_TAG "${GFX_RESEARCH_BASE_GIT_TAG}")
    FetchContent_MakeAvailable(gfx_research_base)
endif()

set(GFX_SHADOWS_ACTUAL_BASE_REVISION "${GFX_RESEARCH_BASE_GIT_TAG}")
if(GFX_RESEARCH_BASE_LOCAL_PATH)
    find_package(Git QUIET)
    set(GFX_SHADOWS_ACTUAL_BASE_REVISION "local-unversioned")
    if(GIT_FOUND)
        execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GFX_RESEARCH_BASE_LOCAL_PATH}" rev-parse HEAD
            OUTPUT_VARIABLE base_commit OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE base_git_result)
        if(base_git_result EQUAL 0)
            set(GFX_SHADOWS_ACTUAL_BASE_REVISION "${base_commit}")
            execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GFX_RESEARCH_BASE_LOCAL_PATH}" diff --quiet
                RESULT_VARIABLE base_dirty ERROR_QUIET)
            if(NOT base_dirty EQUAL 0)
                string(APPEND GFX_SHADOWS_ACTUAL_BASE_REVISION "-dirty")
            endif()
        endif()
    endif()
endif()

add_executable(gfx-research-shadows src/main.cpp src/renderer.cpp src/experiment.cpp src/performance.cpp)
target_include_directories(gfx-research-shadows PRIVATE include)
target_link_libraries(gfx-research-shadows PRIVATE gfx::research_base)
if(UNIX AND NOT APPLE)
    target_sources(gfx-research-shadows PRIVATE src/headless_egl.cpp)
    target_link_libraries(gfx-research-shadows PRIVATE ${CMAKE_DL_LIBS})
    target_compile_definitions(gfx-research-shadows PRIVATE GFX_SHADOWS_HAS_EGL_HEADLESS=1)
endif()
target_compile_definitions(gfx-research-shadows PRIVATE
    GFX_SHADOWS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
    GFX_SHADOWS_BASE_REVISION="${GFX_SHADOWS_ACTUAL_BASE_REVISION}"
    GFX_SHADOWS_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}"
    GFX_SHADOWS_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
if(COMMAND gfx_research_configure_executable)
    gfx_research_configure_executable(gfx-research-shadows)
endif()
if(MSVC)
    target_compile_options(gfx-research-shadows PRIVATE /W4)
else()
    target_compile_options(gfx-research-shadows PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(shadows-geometry-tests tests/test_geometry.cpp)
    target_include_directories(shadows-geometry-tests PRIVATE include)
    target_link_libraries(shadows-geometry-tests PRIVATE gfx_research_glm)
    add_test(NAME shadows-geometry COMMAND shadows-geometry-tests)
    add_executable(shadows-camera-tests tests/test_camera.cpp)
    target_include_directories(shadows-camera-tests PRIVATE include)
    target_link_libraries(shadows-camera-tests PRIVATE gfx::research_base)
    add_test(NAME shadows-camera COMMAND shadows-camera-tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME shadows-analyzer COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_analyzer.py")
    endif()
endif()
